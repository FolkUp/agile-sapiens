// Reading Quality Assessment
function trackReadingQuality() {
  const startTime = Date.now();
  const contentLength = document.querySelector('.content').textContent.length;
  const expectedReadingTime = Math.ceil(contentLength / 200) * 60000; // 200 wpm in ms
  
  // Track deep engagement indicators
  trackFootnoteInteractions();
  trackCitationFollows();
  trackReturnReading();
  
  // Quality completion tracking
  window.addEventListener('beforeunload', () => {
    const actualTime = Date.now() - startTime;
    const completionQuality = (actualTime / expectedReadingTime) * 100;
    
    umami.track('reading-completion', {
      quality_score: Math.min(completionQuality, 100),
      engagement_type: completionQuality > 80 ? 'deep' : 'surface',
      intellectual_indicator: completionQuality > 60
    });
  });
}
