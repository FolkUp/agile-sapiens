{{- with $epub -}} {{- $mb := printf "%.1f" (div (float .Size) 1000000.0) -}} 📖 ePub {{ $mb }} {{ $unit }} {{- end -}} {{- with $pdf -}} {{- $mb := printf "%.1f" (div (float .Size) 1000000.0) -}} 📄 PDF {{ $mb }} {{ $unit }} {{- end -}}
